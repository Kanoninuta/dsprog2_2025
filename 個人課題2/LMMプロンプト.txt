"""
以下のようなコードを書いたんだけど，"module 'flet.controls.material.icons' has no attribute 'LOCATION_ON_OUTLINED'"というエラーが出てしまいました
何が悪い？

import flet as ft
import requests

# 地域リストを取得する関数
def get_area_list():
    area_url = "http://www.jma.go.jp/bosai/common/const/area.json"
    data_json = requests.get(area_url).json()
    return data_json["offices"]

# 選択した地域の天気予報を取得する関数
def get_weather_forecast(area_code):
    forecast_url = f"https://www.jma.go.jp/bosai/forecast/data/forecast/{area_code}.json"
    try:
        data_json = requests.get(forecast_url).json()
        # 天気情報をパースする
        forecast = data_json[0]["timeSeries"][0]["areas"][0]
        area_name = forecast["area"]["name"]
        weather_forecast = forecast["weathers"]
        return area_name, weather_forecast
    except Exception as e:
        return None, [f"エラー: {e}"]

# メインアプリ
def main(page: ft.Page):
    page.title = "気象庁天気予報アプリ"
    page.horizontal_alignment = ft.CrossAxisAlignment.STRETCH

    # 地域リストを取得
    area_list = get_area_list()

    # 選択された地域コード
    selected_area_code = ft.Ref()
    # selected_area_code = ft.Ref[str]("")
    forecast_output = ft.Ref[ft.Column]()

    # 地域リストをNavigationRailに設定
    nav_rail = ft.NavigationRail(
        destinations=[
            ft.NavigationRailDestination(label=ft.Text(area["name"]), icon=ft.icons.WB_SUNNY)
            for area in area_list
        ],
        selected_index=0,
        on_change=lambda e: show_forecast(e.control.selected_index),
    )

    # 天気予報を表示する関数
    def show_forecast(index):
        selected_area = area_list[index]
        selected_area_code.current = selected_area["code"]
        area_name, forecast_list = get_weather_forecast(selected_area_code.current)

        forecast_output.current.controls = [
            ft.Text(f"{area_name}の天気予報:", size=20, weight=ft.FontWeight.BOLD)
        ]
        for forecast in forecast_list:
            forecast_output.current.controls.append(ft.Text(forecast))
        forecast_output.current.update()

    # 初期表示
    show_forecast(0)

    # UIを構築
    page.add(
        ft.Row(
            [
                nav_rail,
                ft.VerticalDivider(width=1),
                ft.Column(
                    ref=forecast_output,
                    expand=True,
                ),
            ]
        )
    )

ft.app(target=main)
"""

👉 ft.icons.LOCATION_ON_OUTLINED というアイコンは存在しない

🔎 まず確認してほしいこと

エラーメッセージはこうだよね：

module 'flet.controls.material.icons' has no attribute 'LOCATION_ON_OUTLINED'


つまり

そんな名前のアイコン定数は定義されてません

って言われてる。

🎯 なぜ起きる？

Fletの icons は
Material Icons に対応してるけど、

✔ 使えるアイコン名は「厳密一致」

例えば：

ft.icons.LOCATION_ON


はあるけど

ft.icons.LOCATION_ON_OUTLINED


はない可能性が高い。

✅ 解決方法
方法①：存在するアイコンに変える（おすすめ）

たとえば：

ft.icons.LOCATION_ON


または

ft.icons.MAP
ft.icons.PLACE
ft.icons.WB_SUNNY


など。

方法②：文字列で直接指定する

Fletは文字列でも指定できる：

icon="location_on_outlined"


ただしこれはFletのバージョン依存。

💡 今のあなたのコードを見ると…

ここでは：

icon=ft.icons.WB_SUNNY


になってるよね？

だから実は、
どこか別の場所で LOCATION_ON_OUTLINED を使ってる可能性が高い。

🔍 チェックしてほしい

ファイル内検索で
LOCATION_ON_OUTLINED

別の.pyファイルで使ってない？

Fletのバージョンは最新？

🧠 ついでにワンポイント（Kanon向け）

Fletのアイコンは：

dir(ft.icons)


で一覧見れるよ。

これを一回 print して確認するのが一番確実。

✨ まとめ

原因：

存在しないアイコン名を指定している

対処：

正しいアイコン名に修正する

もしよかったら
pip show flet のバージョン教えて。

そこまで一緒に整えるよ 😎
